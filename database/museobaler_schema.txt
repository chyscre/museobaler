====================================================================
 MUSEOBALER DATABASE - TABLE REFERENCE
====================================================================

CONTENTS
  VISITORS & ATTENDANCE    visitors, visit_groups, attendances, attendance_corrections, scans, tours
  STAFF                    staff, staff_attendances, staff_attendance_days, staff_schedules
  MUSEUM CONTENT           exhibits, exhibit_images, exhibit_translations, categories, museum_halls, museum_info, bookmarks
  AUTH & APP               users, sessions, password_reset_tokens, notifications, feedback, logs

####################################################################
#  VISITORS & ATTENDANCE
####################################################################

VISITORS
--------------------------------------------------------------------
  visitor_id      bigint unsigned                     PRIMARY KEY, auto
  group_id        bigint unsigned                     optional, -> visit_groups.group_id (on delete set null)
  first_name      varchar(255)                        required
  last_name       varchar(255)                        required
  middle_name     varchar(255)                        optional
  age             int                                 optional
  sex             enum('Male','Female','Other','Prefer not to say')
                                                      optional
  visitor_type    enum('Local','Tourist','Foreign')   required, default Local
  admission_fee   decimal(8,2)                        required, default 0.00
  payment_status  enum('Free','Unpaid','Paid')        required, default Free
  paid_at         timestamp                           optional
  id_verified     tinyint(1)                          required, default 0
  verified_at     timestamp                           optional
  verified_by     varchar(255)                        optional
  visit_type      enum('Solo','Group','School','Family','Walk-in')
                                                      required, default Solo
  city            varchar(255)                        optional
  province        varchar(255)                        optional
  country         varchar(255)                        optional, default Philippines
  email           varchar(255)                        optional
  auth_provider   varchar(255)                        required, default manual
  source          enum('app','kiosk','desk','recovered')
                                                      required, default app
  registered_by   bigint unsigned                     optional
  explore_mode    varchar(255)                        required, default Storyline
  last_visit      timestamp                           optional
  created_at      timestamp                           optional
  updated_at      timestamp                           optional
  ..................................................................
  index    group_id
  index    created_at

VISIT_GROUPS
--------------------------------------------------------------------
  group_id        bigint unsigned                     PRIMARY KEY, auto
  group_name      varchar(255)                        optional
  group_type      enum('Family','Group','School','Tour')
                                                      required, default Group
  contact_name    varchar(255)                        optional
  contact_phone   varchar(255)                        optional
  visitor_type    enum('Local','Tourist','Foreign')   required, default Tourist
  city            varchar(255)                        optional
  province        varchar(255)                        optional
  country         varchar(255)                        optional, default Philippines
  headcount       int unsigned                        required, default 1
  paying_count    int unsigned                        required, default 0
  total_fee       decimal(10,2)                       required, default 0.00
  payment_status  enum('Free','Unpaid','Paid')        required, default Unpaid
  paid_at         timestamp                           optional
  visit_date      date                                required
  registered_by   bigint unsigned                     optional
  notes           text                                optional
  created_at      timestamp                           optional
  updated_at      timestamp                           optional
  ..................................................................
  index    visit_date
  index    payment_status

ATTENDANCES
--------------------------------------------------------------------
  attendance_id  bigint unsigned  PRIMARY KEY, auto
  visitor_id     bigint unsigned  optional, -> visitors.visitor_id (on delete set null)
  visitor_name   varchar(255)     optional
  latitude       decimal(10,7)    optional
  longitude      decimal(10,7)    optional
  accuracy       int              optional
  method         varchar(255)     required, default geofence
  visit_date     date             required
  created_at     timestamp        optional
  updated_at     timestamp        optional
  exited_at      timestamp        optional
  duration_mins  int              optional
  ..................................................................
  unique   visitor_id, visit_date

ATTENDANCE_CORRECTIONS
--------------------------------------------------------------------
  correction_id   bigint unsigned                     PRIMARY KEY, auto
  staff_id        bigint unsigned                     required, -> staff.staff_id (on delete cascade)
  work_date       date                                required
  type            enum('in','out')                    required
  requested_time  time                                required
  reason          text                                required
  requested_by    bigint unsigned                     required
  requested_at    timestamp                           required, default CURRENT_TIMESTAMP
  status          enum('Pending','Approved','Rejected')
                                                      required, default Pending
  reviewed_by     bigint unsigned                     optional
  reviewed_at     timestamp                           optional
  review_note     text                                optional
  created_at      timestamp                           optional
  updated_at      timestamp                           optional
  ..................................................................
  index    staff_id
  index    status, work_date

SCANS
--------------------------------------------------------------------
  scan_id        bigint unsigned     PRIMARY KEY, auto
  exhibit_id     bigint unsigned     required, -> exhibits.exhibit_id (on delete cascade)
  visitor_id     bigint unsigned     optional, -> visitors.visitor_id (on delete set null)
  language_code  varchar(10)         optional
  scan_type      enum('qr','image')  optional
  scanned_at     timestamp           required, default CURRENT_TIMESTAMP
  created_at     timestamp           optional
  updated_at     timestamp           optional
  ..................................................................
  index    exhibit_id

TOURS

  tour_id         bigint unsigned                     PRIMARY KEY, auto

STAFF
--------------------------------------------------------------------
  staff_id        bigint unsigned                     PRIMARY KEY, auto
  name            varchar(255)                        required
  email           varchar(255)                        required
  password        varchar(255)                        required
  role            enum('SuperAdmin','Administrator')  required, default Administrator
  status          tinyint(1)                          required, default 1
  remember_token  varchar(100)                        optional
  created_at      timestamp                           optional
  updated_at      timestamp                           optional
  ..................................................................

STAFF_ATTENDANCES
--------------------------------------------------------------------
  staff_attendance_id  bigint unsigned      PRIMARY KEY, auto
  staff_id             bigint unsigned      required, -> staff.staff_id (on delete cascade)
  work_date            date                 required
  type                 enum('in','out')     required
  scanned_at           timestamp            required, default CURRENT_TIMESTAMP
  latitude             decimal(10,7)        optional
  longitude            decimal(10,7)        optional
  accuracy             int                  optional
  distance_m           int                  optional
  method               enum('qr','manual')  required, default qr
  recorded_by          bigint unsigned      optional
  ip_address           varchar(255)         optional
  user_agent           varchar(512)         optional
  created_at           timestamp            optional
  updated_at           timestamp            optional
  ..................................................................
  unique   staff_id, work_date, type
  index    work_date

STAFF_ATTENDANCE_DAYS
--------------------------------------------------------------------
  work_date   date             PRIMARY KEY
  day_secret  varchar(64)      required
  opened_by   bigint unsigned  optional
  created_at  timestamp        optional
  updated_at  timestamp        optional

STAFF_SCHEDULES

  shift_end      time               required
  grace_minutes  smallint unsigned  required, default 15
  is_rest_day    tinyint(1)         required, default 0
  created_at     timestamp          optional
  updated_at     timestamp          optional
  ..................................................................
  unique   staff_id, weekday

EXHIBITS
--------------------------------------------------------------------
  exhibit_id       bigint unsigned                     PRIMARY KEY, auto
  exhibit_code     varchar(255)                        required
  name             varchar(255)                        required

  fun_facts        text                                optional
  category_id      bigint unsigned                     optional, -> categories.category_id (on delete set null)
  floor            enum('Ground Floor','2nd Floor')    required, default Ground Floor
  hall             enum('Hall A','Hall B','Hall C','Hall D','Hall E')
                                                       optional
  authors          varchar(255)                        optional
  languages        varchar(255)                        required, default Filipino,English
  storyline_order  int                                 required, default 0
  image            varchar(255)                        optional
  status           tinyint(1)                          required, default 1
  date_published   date                                optional
  created_at       timestamp                           optional
  updated_at       timestamp                           optional
  ..................................................................
  unique   exhibit_code
  index    category_id
